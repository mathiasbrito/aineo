--- Lets the child idle, and so redraw, for `ms`.
local function idle(ms)
  vim.wait(ms, function()

  end, 50)

--- The last `stty size` the terminal job of `buffer` printed.
local function last_size(buffer)
  return child.lua(
    [[
    local sizes = vim.tbl_filter(function(line)
      return line:match('^%d+ %d+$') ~= nil
    end, vim.api.nvim_buf_get_lines(..., 0, -1, false))
    return sizes[#sizes] or 'none'
  ]],
    { buffer }
  )
end

--- Starts the size loop in a buffer shown in two windows of `first` and
--- `second` rows, the second in another tab page when `other_tab` is true.
local function shown_twice(first, second, other_tab, redrawn)
  return child.lua(
    [[
    local command, first, second, other_tab, redrawn = ...
    vim.cmd('enew')
    local buffer = vim.api.nvim_get_current_buf()
    if other_tab then
      vim.cmd('tab split')
      vim.api.nvim_win_set_height(0, second)
      vim.cmd('split')
      vim.api.nvim_win_set_height(0, second)
      vim.cmd('tabprevious')
      vim.cmd('split')
      vim.api.nvim_win_set_height(0, first)
    else
      vim.cmd('split')
      vim.api.nvim_win_set_height(0, first)
    end
    if redrawn then
      _G.t30_redraws = vim.uv.new_timer()
      _G.t30_redraws:start(0, 50, vim.schedule_wrap(function() vim.cmd('redraw') end))
    end
    vim.fn.jobstart(command, { term = true })
    local heights = vim.tbl_map(function(window)
      return vim.api.nvim_tabpage_get_number(vim.api.nvim_win_get_tabpage(window)) .. ':' .. vim.api.nvim_win_get_height(window)
    end, vim.fn.win_findbuf(buffer))
    return { buffer = buffer, heights = table.concat(heights, ' ') }
  ]],
    { SIZE_LOOP, first, second, other_tab, redrawn }
  )
end

--- code: t30-probe-hangup.lua
-- Probe C5: what Neovim does to a terminal job after `jobstop()` hangs it up,
-- for a process that ignores nothing, one that ignores only SIGHUP, and one
-- that ignores SIGHUP and SIGTERM: how long until it exits, and its code.
local COMMANDS = {
  { 'ignores nothing', 'while :; do sleep 0.1; done' },
  { 'ignores SIGHUP', 'trap "" HUP; while :; do sleep 0.1; done' },
  { 'ignores SIGHUP and SIGTERM', 'trap "" HUP TERM; while :; do sleep 0.1; done' },
}

for _, entry in ipairs(COMMANDS) do
  local name, script = entry[1], entry[2]
  vim.cmd('enew')
  local exited_at, code
  local job = vim.fn.jobstart({ 'sh', '-c', script }, {
    term = true,
    on_exit = function(_, exit_code)
      exited_at, code = vim.uv.hrtime(), exit_code
    end,
  })
  vim.wait(500, function()
    return false
  end)
  local stopped_at = vim.uv.hrtime()
  vim.fn.jobstop(job)
  vim.wait(8000, function()
    return exited_at ~= nil
  end, 10)
  print(
    ('%-28s exited %s s after jobstop(), code %s'):format(
      name,
      exited_at and ('%.2f'):format((exited_at - stopped_at) / 1e9) or 'never (8 s)',
      tostring(code)
    )
  )
end
print(vim.fn.execute('version'):match('NVIM v[^\n]*'))

-- Probe C6: whether `vim.wait()`'s time-out runs out while a child writes
-- without end (`yes`), with every chunk read by a callback, as the health
-- check's command does; and `SystemObj:wait()` under the same flood.
local function flood(no_op)
  local bytes = 0
  local process = vim.system({ 'yes', 'flood' }, {
    stdout = no_op and function() end or function(_, data)
      bytes = bytes + #(data or '')
    end,
  })
  local watchdog = assert(vim.uv.new_timer())
  watchdog:start(10000, 0, function()
    process:kill(9)
    io.stdout:write('  (watchdog: yes killed 10 s after it started)\n')
    io.stdout:flush()
  end)
  return process, function()
    watchdog:stop()
    watchdog:close()
    return bytes
  end
end

local function measure(name, wait, no_op)
  local process, bytes = flood(no_op)
  local started = vim.uv.hrtime()
  wait(process)
  local elapsed = (vim.uv.hrtime() - started) / 1e6
  process:kill(9)
  process:wait(2000)
  io.stdout:write(('%-34s returned after %d ms, %d bytes read\n'):format(name, elapsed, bytes()))
  io.stdout:flush()
end

measure('vim.wait(1000, false)', function()
  vim.wait(1000, function()
    return false
  end)
end)
measure('vim.wait(1000, false, 200, true)', function()
  vim.wait(1000, function()
    return false
  end, 200, true)
end)
measure('SystemObj:wait(1000)', function(process)
  process:wait(1000)
end)
measure('vim.wait(1000, false), no-op handler', function()
  vim.wait(1000, function()
    return false
  end)
end, true)
measure('vim.wait(1000, false, 10, true), no-op', function()
  vim.wait(1000, function()
    return false
  end, 10, true)
end, true)
print(vim.fn.execute('version'):match('NVIM v[^\n]*'))

Run: t30fix-run-flood-reach.sh (Neovim v0.12.5), each variant twice, a 3 s watchdog killing the writer by pid.
The brief review of wave 7's T24 packet (T30-2) reported that on 0.12.5 vim.wait(1000, …) returned after 1002 ms under a `yes` flood (1003 ms with fast_only); its probe file was not kept. That figure is what a flood that never reaches Neovim gives: with no pipe opened for the writer's output (stdout = false), or with the writer sending it to /dev/null itself, the wait returns at 1 s while the writer still floods. A flood read by a handler, as run_within_bound() reads claude.cmd --version, keeps the wait from running out until the writer is killed. PR #108's attack review measured the wider matrix (finding 1): 1002 ms only for stdout = false, a writer sending to /dev/null, and a handler that spends 1 ms a chunk; about 4000 ms — the writer killed at 3 s — for every handler that reads the flood cheaply, the health check's among them. The wait still subtracts whole milliseconds only (src/nvim/event/multiqueue.h LOOP_PROCESS_EVENTS_UNTIL, the same body in v0.11.6 and v0.12.5, by the attack review's diff of the two), so a loop whose passes each take under 1 ms never runs it out.

--- code: t30fix-probe-flood-reach.lua
-- T30 fix round, probe C6b: whether `vim.wait(1000, …)` runs out at 1 s while
-- a writer floods, by whether the flood reaches Neovim's loop. One variant
-- per Neovim (`nvim --clean --headless -l t30fix-probe-flood-reach.lua
-- <variant>`); a timer kills the writer by pid 3 s after it starts. Prints
-- how long the wait took and whether the writer still ran when it returned.
local variant = arg[1]
local WATCHDOG_MS = 3000

local writers = {
  -- read by a handler, as run_within_bound() reads `claude.cmd --version`
  read_by_a_handler = function()
    local kept = 0
    return vim.system({ 'yes', 'flood' }, {
      stdout = function(_, data)
        kept = kept + #(data or '')
      end,
    })
  end,
  -- not read at all: Neovim opens no pipe for its standard output
  stdout_false = function()
    return vim.system({ 'yes', 'flood' }, { stdout = false })
  end,
  -- written to /dev/null by the writer itself: nothing reaches the pipe
  to_dev_null = function()
    return vim.system({ 'sh', '-c', 'exec yes flood >/dev/null' }, { stdout = function() end })
  end,
}

local process = writers[variant]()
local killed_at
local watchdog = assert(vim.uv.new_timer())
watchdog:start(WATCHDOG_MS, 0, function()
  killed_at = vim.uv.hrtime()
  vim.uv.kill(process.pid, 'sigkill')
end)
local started = vim.uv.hrtime()
vim.wait(1000, function()
  return false
end)
local returned = vim.uv.hrtime()
local still_flooding = killed_at == nil
watchdog:stop()
watchdog:close()
vim.uv.kill(process.pid, 'sigkill')
io.stdout:write(
  ('%-18s vim.wait(1000) returned after %5d ms; the writer %s\n'):format(
    variant,
    (returned - started) / 1e6,
    still_flooding and 'still flooding' or ('killed by the watchdog at %d ms'):format((killed_at - started) / 1e6)
  )
)
io.stdout:flush()
