T26 (Visual Send) — the probes behind its brief, D20's "measurement first"
==========================================================================

Measured 2026-10-05 on macOS arm64 (Darwin 27.0.0), NVIM v0.12.5 (Homebrew,
LuaJIT 2.1.1788856981). Nothing ran on 0.11. The code measured is T24's head,
8e520f5 (origin/feature/t24-panes, PR #105), extracted with `git archive` into
a scratch folder; every probe loads it through its own scripts/minimal_init.lua,
so tests/helpers/entry_guard/ is first on PATH and the real `claude` never ran.
Claude Code is the suites' fake (tests/helpers/fake_claude.lua, mode `ready`),
run as aineo's `claude.cmd`. Each probe ran headless in a home of its own
(run.sh: XDG_*, CLAUDE_CONFIG_DIR, NVIM_LOG_FILE and TMPDIR under it), and
quit by itself; no process was left.

Keys are typed with nvim_feedkeys(…, 'mtx'): typed, remapped, executed at once,
so each key sequence is its own command and its own undo block, as typed keys
are. `\s` is the prefix key aineo maps at VimEnter (<Plug>(aineo-send)).

What they show, in short (the outputs below are the record):
- U1 `u` in Input after a whole-Input Send brings Input's text back; <C-r>
  empties it again. One `u` per Send (U7). Sent from another window (U2), or
  while the changes pane hides Input (U3), `u` in Input still brings it back.
  A Send from Insert mode through <C-o> is its own undo step (U6).
- U4 a failed write (the write raising, as a closed stream does): Send's
  removal and its put-back are ONE undo block, so the first `u` after it
  changes nothing visible; the second undoes the user's own last change.
- U5 a refused Send changes nothing and adds no undo step.
- D the draft restored into Input (D17, `undolevels = -1`) is not undone by
  `u` ("Already at oldest change"); after a Send, `u` brings the draft's text
  back, and the draft file holds it again a second later; <C-r> empties the
  draft file at once.
- V for every selection kind, Vim's own `"_d` and `:normal! "_d` inside an
  x mapping's Lua callback remove the same text, and one `u` restores it.
  getregion() gives Vim's own yank text in every case but one: a selection
  ending past the end of a line (`v$`), where the yank ends with an empty
  line (the line break) and getregion() does not — yet `"_d` removes the
  line break, joining the lines.
- V in an x mapping's Lua callback, Visual mode is still on and '< '> still
  hold the PREVIOUS selection: the selection is read from getpos('v') and
  getpos('.'), or after leaving Visual mode.
- V `\s` typed in Visual mode where nothing maps it: `\` does nothing and `s`
  replaces the selection — the selected text is deleted and Insert mode
  entered.
- V2 a blockwise removal made of one nvim_buf_set_text() per line in one
  callback is one undo step; a callback that changes nothing leaves Visual
  mode on; `:'<,'>Aineo send` and `:1Aineo send` raise E481 (no range
  allowed) on T24's `:Aineo`.

==================== run.sh ====================
#!/bin/sh
# Runs one probe headless on the newest Neovim, isolated under this folder.
# usage: run.sh <probe.lua> <home-name> [extra nvim args...]
HERE=$(cd "$(dirname "$0")" && pwd)
TREE=$(cd "$HERE/../../t24tree" && pwd)
PROBE=$1; NAME=$2; shift 2
H="$HERE/home-$NAME"
rm -rf "$H"; mkdir -p "$H/config" "$H/data" "$H/state" "$H/cache" "$H/claude" "$H/tmp" "$H/work"
for v in $(env | sed -n 's/^\(CLAUDE[A-Z_]*\)=.*/\1/p'); do unset "$v"; done
unset AINEO_CHILD
export XDG_CONFIG_HOME="$H/config" XDG_DATA_HOME="$H/data" XDG_STATE_HOME="$H/state" XDG_CACHE_HOME="$H/cache"
export CLAUDE_CONFIG_DIR="$H/claude" NVIM_LOG_FILE="$H/nvim.log" TMPDIR="$H/tmp"
export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 GIT_CEILING_DIRECTORIES="$HERE"
export GIT_AUTHOR_NAME=probe GIT_AUTHOR_EMAIL=probe@example.invalid GIT_COMMITTER_NAME=probe GIT_COMMITTER_EMAIL=probe@example.invalid
export AINEO_PROBE_TREE="$TREE" AINEO_PROBE_HOME="$H" AINEO_PROBE_OUT="$HERE/$NAME.out"
cd "$H/work" || exit 1
exec nvim --headless -u "$TREE/scripts/minimal_init.lua" "$@" -c "luafile $HERE/$PROBE"

-- Shared helpers for the T25/T26 brief probes.
local M = {}
local out = {}

function M.log(...)
  local parts = {}
  for i = 1, select('#', ...) do
    local v = select(i, ...)
    parts[#parts + 1] = type(v) == 'string' and v or vim.inspect(v)
  end
  out[#out + 1] = table.concat(parts, ' ')
end

function M.finish()
  vim.fn.writefile(out, vim.env.AINEO_PROBE_OUT)
  vim.cmd('qall!')
end

-- No probe outlives 80 s: it writes what it has and quits.
vim.defer_fn(function()
  M.log('TIMEOUT: the probe ran past its limit')
  M.finish()
end, tonumber(vim.env.AINEO_PROBE_LIMIT_MS) or 80000)

function M.keys(k)
  vim.api.nvim_feedkeys(vim.api.nvim_replace_termcodes(k, true, false, true), 'mtx', false)
end

function M.lines(buf)
  return vim.api.nvim_buf_get_lines(buf, 0, -1, false)
end

M.messages = {}
vim.notify = function(message, level)
  table.insert(M.messages, { message = message, level = level })
end

function M.take_messages()
  local m = M.messages
  M.messages = {}
  return m
end

--- Runs `fn` once the editor has started and aineo's VimEnter handler has
--- mapped the prefix, then writes the log and quits.
function M.main(fn)
  local function run()
    vim.schedule(function()
      local ok, err = xpcall(fn, debug.traceback)
      if not ok then
        M.log('PROBE ERROR:', err)
      end
      M.finish()
    end)
  end
  if vim.v.vim_did_enter == 1 then
    run()
  else
    vim.api.nvim_create_autocmd('VimEnter', { once = true, callback = run })
  end
end

--- Points aineo at the suites' fake claude in ready mode.
function M.use_fake(mode)
  local tree = vim.env.AINEO_PROBE_TREE
  vim.env.AINEO_FAKE_CLAUDE_RECORD = vim.env.AINEO_PROBE_HOME .. '/record.jsonl'
  vim.env.AINEO_FAKE_CLAUDE_MODE = mode or 'ready'
  vim.g.aineo = {
    autostart = false,
    claude = { cmd = { vim.v.progpath, '--clean', '-l', tree .. '/tests/helpers/fake_claude.lua' } },
  }
end

function M.received()
  local ok, lines = pcall(vim.fn.readfile, vim.env.AINEO_FAKE_CLAUDE_RECORD)
  if not ok then
    return {}
  end
  local got = {}
  for _, line in ipairs(lines) do
    local decoded = vim.json.decode(line)
    if decoded.received then
      got[#got + 1] = decoded.received
    end
  end
  return got
end

function M.wait_ready()
  return vim.wait(15000, function()
    return require('aineo.claude').session_status() == 'ready'
  end, 50)
end

return M

-- T26 probe V: what a Visual selection in aineo's real Input gives
-- getregion(), what Vim's own "_d removes, and how many `u` bring it back,
-- for each kind of selection. Run with run.sh p_visual.lua visual.
local P = dofile(vim.fn.fnamemodify(debug.getinfo(1, 'S').source:sub(2), ':h') .. '/lib.lua')

local CTRL_V = vim.api.nvim_replace_termcodes('<C-v>', true, false, true)

local CASES = {
  { 'charwise, one line, multibyte', { 'héllo wörld' }, '0fwv3l' },
  { 'charwise, two lines', { 'alpha beta', 'gamma delta' }, '0wvj' },
  { 'charwise, ending past the line (v$)', { 'one', 'two', 'three' }, '0lv$' },
  { 'charwise, v$ then j', { 'one', 'two', 'three' }, '0lv$j' },
  { 'linewise, one line', { 'a1', 'a2', 'a3' }, 'jV' },
  { 'linewise, to the end', { 'a1', 'a2', 'a3' }, 'jVG' },
  { 'blockwise, ragged lines', { 'abcdef', 'ab', 'abcdef' }, '0ll<C-v>jjl' },
  { 'blockwise, $', { 'abcdef', 'abc', 'abcdefgh' }, '0l<C-v>jj$' },
  { "charwise, 'selection' exclusive", { 'hello world' }, '0v4l', { selection = 'exclusive' } },
  { "blockwise, 'selection' exclusive", { 'abcdef', 'abcdef' }, '0l<C-v>jll', { selection = 'exclusive' } },
  { 'blockwise, a double-width character cut', { 'a漢字b', 'abcdef' }, '0l<C-v>j' },
  { 'blockwise, double-width whole', { 'a漢字b', 'abcdef' }, '0l<C-v>jl' },
  { 'charwise, backwards', { 'hello world' }, '$vb' },
  { 'charwise, ending on a multibyte character', { 'aé b' }, '0vl' },
  { 'linewise, the whole buffer', { 'x', 'y' }, 'ggVG' },
  { 'linewise, an empty line', { '', 'x' }, 'V' },
  { 'charwise, a tab', { 'a\tb' }, '0lv' },
}

local probe_result

--- The mapping's callback: what a Visual-mode `\s` sees in Visual mode.
local function seen_in_visual()
  local mode = vim.fn.mode()
  local from, to = vim.fn.getpos('v'), vim.fn.getpos('.')
  local region = vim.fn.getregion(from, to, { type = mode })
  local regionpos = vim.fn.getregionpos(from, to, { type = mode })
  local marks = { vim.fn.getpos("'<"), vim.fn.getpos("'>") }
  local curswant = vim.fn.winsaveview().curswant
  -- Vim's own yank of the same selection, into register z, kept and restored.
  local saved = vim.fn.getreginfo('z')
  vim.cmd('normal! "zy')
  local yank = vim.fn.getreg('z', 1, true)
  local yank_type = vim.fn.getregtype('z')
  vim.fn.setreg('z', saved)
  probe_result = {
    mode = mode,
    region = region,
    regionpos = regionpos,
    marks_before_exit = marks,
    curswant = curswant,
    yank = yank,
    yank_type = yank_type,
  }
end

P.main(function()
  P.log('nvim', tostring(vim.version()))
  local layout = require('aineo.layout')
  layout.open({
    claude = vim.api.nvim_create_buf(false, true),
    report = vim.api.nvim_create_buf(false, true),
    report_height = 2 / 3,
  })
  local input = layout.input_buffer()
  vim.keymap.set('x', '<F9>', seen_in_visual, { buffer = input })
  for _, case in ipairs(CASES) do
    local name, lines, select_keys, options = case[1], case[2], case[3], case[4] or {}
    local saved_selection = vim.o.selection
    if options.selection then
      vim.o.selection = options.selection
    end
    vim.api.nvim_set_current_win(vim.fn.bufwinid(input))
    -- 1: what the mapping sees.
    vim.api.nvim_buf_set_lines(input, 0, -1, false, lines)
    P.keys('<Esc>gg')
    P.keys(select_keys .. '<F9>')
    P.keys('<Esc>')
    local seen = probe_result
    -- 2: Vim's own "_d of the same selection, then `u` until the text is back.
    vim.api.nvim_buf_set_lines(input, 0, -1, false, lines)
    P.keys('<Esc>gg')
    local seq_before = vim.fn.undotree(input).seq_cur
    P.keys(select_keys .. '"_d')
    local after_d = P.lines(input)
    local cursor_after_d = vim.api.nvim_win_get_cursor(0)
    local undos = 0
    while not vim.deep_equal(P.lines(input), lines) and undos < 5 do
      P.keys('u')
      undos = undos + 1
    end
    -- 3: the same removal from inside an x mapping's Lua callback: :normal! "_d.
    vim.keymap.set('x', '<F8>', function()
      vim.cmd('normal! "_d')
    end, { buffer = input })
    vim.api.nvim_buf_set_lines(input, 0, -1, false, lines)
    P.keys('<Esc>gg')
    P.keys(select_keys .. '<F8>')
    local after_mapping = P.lines(input)
    local mode_after_mapping = vim.fn.mode()
    local mapping_undos = 0
    while not vim.deep_equal(P.lines(input), lines) and mapping_undos < 5 do
      P.keys('u')
      mapping_undos = mapping_undos + 1
    end
    vim.o.selection = saved_selection
    P.log('=== V: ' .. name, '— lines', lines, 'keys', select_keys, 'selection', options.selection or 'inclusive')
    P.log('  mode in the callback:', seen.mode, ' curswant:', seen.curswant)
    P.log('  getregion:', seen.region)
    P.log('  Vim yank of it:', seen.yank, 'type', seen.yank_type)
    P.log('  getregion == yank:', vim.deep_equal(seen.region, seen.yank))
    P.log('  joined by line feeds:', table.concat(seen.region, '\n'))
    P.log('  getregionpos:', vim.tbl_map(function(pair)
      return { pair[1][2], pair[1][3], pair[1][4], '..', pair[2][2], pair[2][3], pair[2][4] }
    end, seen.regionpos))
    P.log("  '< '> while still in Visual mode:", seen.marks_before_exit[1][2], seen.marks_before_exit[1][3], seen.marks_before_exit[2][2], seen.marks_before_exit[2][3])
    P.log('  after "_d:', after_d, 'cursor', cursor_after_d, ' u to restore:', undos)
    P.log('  after :normal! "_d in an x mapping:', after_mapping, 'mode after:', mode_after_mapping, ' u to restore:', mapping_undos)
  end
  -- What `\s` typed in Visual mode does where nothing maps it.
  local plain = vim.api.nvim_create_buf(true, false)
  vim.api.nvim_set_current_buf(plain)
  vim.api.nvim_buf_set_lines(plain, 0, -1, false, { 'keep this text' })
  P.keys('<Esc>gg0vll\\s')

-- T26 probe V2: removal by API calls inside an x mapping, counted in `u`;
-- `:'<,'>Aineo send` on T24's code; a Visual-mode key's callback after a
-- refusal. Run with run.sh p_visual2.lua visual2.
local P = dofile(vim.fn.fnamemodify(debug.getinfo(1, 'S').source:sub(2), ':h') .. '/lib.lua')

P.main(function()
  P.log('nvim', tostring(vim.version()))
  local layout = require('aineo.layout')
  layout.open({
    claude = vim.api.nvim_create_buf(false, true),
    report = vim.api.nvim_create_buf(false, true),
    report_height = 2 / 3,
  })
  local input = layout.input_buffer()
  vim.api.nvim_set_current_win(vim.fn.bufwinid(input))

  -- A blockwise removal made of one nvim_buf_set_text() per line, bottom up.
  vim.keymap.set('x', '<F7>', function()
    local from, to = vim.fn.getpos('v'), vim.fn.getpos('.')
    local spans = vim.fn.getregionpos(from, to, { type = vim.fn.mode() })
    for index = #spans, 1, -1 do
      local first, last = spans[index][1], spans[index][2]
      if first[3] > 0 then
        vim.api.nvim_buf_set_text(input, first[2] - 1, first[3] - 1, last[2] - 1, last[3], {})
      end
    end
    vim.api.nvim_feedkeys(vim.api.nvim_replace_termcodes('<Esc>', true, false, true), 'n', false)
  end, { buffer = input })
  local lines = { 'abcdef', 'ab', 'abcdef', 'abcdef' }
  vim.api.nvim_buf_set_lines(input, 0, -1, false, lines)
  P.keys('<Esc>gg0ll<C-v>jjjl<F7>')
  P.log('V2a after per-line set_text:', P.lines(input), 'mode', vim.fn.mode())
  local undos = 0
  while not vim.deep_equal(P.lines(input), lines) and undos < 6 do
    P.keys('u')
    undos = undos + 1
  end
  P.log('V2a u to restore:', undos, P.lines(input))

  -- The same from the composition-root way: a function run through pcall, the
  -- removal then an error, then the lines put back with set_lines.
  vim.keymap.set('x', '<F6>', function()
    local from, to = vim.fn.getpos('v'), vim.fn.getpos('.')
    local before = vim.api.nvim_buf_get_lines(input, 0, -1, false)
    vim.cmd('normal! "_d')
    vim.api.nvim_buf_set_lines(input, 0, -1, false, before)
  end, { buffer = input })
  vim.api.nvim_buf_set_lines(input, 0, -1, false, { 'one two three' })
  local seq = vim.fn.undotree(input).seq_cur
  P.keys('<Esc>gg0wviw<F6>')
  P.log('V2b removed then put back in one callback:', P.lines(input), 'seq before', seq, 'after', vim.fn.undotree(input).seq_cur)
  P.keys('u')
  P.log('V2b after u:', P.lines(input), 'seq', vim.fn.undotree(input).seq_cur)

  -- :'<,'>Aineo send on T24's :Aineo (no range declared).
  vim.api.nvim_buf_set_lines(input, 0, -1, false, { 'l1', 'l2' })
  P.keys('<Esc>ggVj')
  P.keys('<Esc>')
  local ok, err = pcall(vim.cmd, "'<,'>Aineo send")
  P.log("V2c :'<,'>Aineo send:", ok, err, 'messages', P.take_messages(), 'Input', P.lines(input))
  local ok2, err2 = pcall(vim.cmd, '1Aineo send')
  P.log('V2c :1Aineo send:', ok2, err2)

  -- What a Lua x mapping leaves for the mode when it returns without leaving Visual.
  vim.keymap.set('x', '<F5>', function()
    vim.notify('refused')
  end, { buffer = input })
  P.keys('<Esc>ggv<F5>')
  P.log('V2d mode after a callback that changes nothing:', vim.fn.mode(), P.take_messages())
  P.keys('<Esc>')
  P.log("V2d '< '> after leaving Visual:", vim.fn.getpos("'<"), vim.fn.getpos("'>"))
end)
