local function show(...) io.stdout:write(table.concat(vim.tbl_map(vim.inspect, { ... }), '  '), '\n') end

-- 1. a cluster: str_utf_end against charidx/byteidx (the attack review, finding 1)
for _, s in ipairs({ 'a👨‍👩b', 'a👍🏽b', 'a🇫🇷b', 'a漢́b', 'a漢b' }) do
  local first = 2 -- byte of the cluster's start (1-based)
  local by_codepoint = first + vim.str_utf_end(s, first)
  local by_char = vim.fn.byteidx(s, vim.fn.charidx(s, first - 1) + 1)
  show('cluster', s, 'str_utf_end ends at', by_codepoint, 'byteidx(charidx) ends at', by_char)
end

-- 2. getregionpos on a block edge cutting the cluster (the attack review, finding 1)
vim.api.nvim_buf_set_lines(0, 0, -1, false, { 'abcdef', 'a👍🏽b', 'abcdef' })
vim.cmd('normal! gg0l\22jj')
local regions = vim.fn.getregionpos(vim.fn.getpos('v'), vim.fn.getpos('.'), { type = '\22' })

vim.cmd('normal! \27')

-- 6. a removal and its put-back in one typed command (the undo Learning; PR #118's records review, finding 7)
for _, broken in ipairs({ false, true }) do
  vim.keymap.set('n', '<F2>', function()
    local held = vim.api.nvim_buf_get_lines(0, 0, -1, false)
    vim.api.nvim_buf_set_lines(0, 0, -1, false, {})
    if broken then vim.cmd('let &g:undolevels = &g:undolevels') end
    vim.api.nvim_buf_set_lines(0, 0, -1, false, held)
  end)
  vim.cmd('enew!')
  vim.api.nvim_feedkeys(vim.keycode('itext<Esc>'), 'mtx', false)
  vim.api.nvim_feedkeys(vim.keycode('<F2>'), 'mtx', false)
  vim.api.nvim_feedkeys('u', 'mtx', false); local u1 = vim.api.nvim_buf_get_lines(0, 0, -1, false)
  vim.api.nvim_feedkeys('u', 'mtx', false)
  show('undo break', broken, 'u', u1, 'u u', vim.api.nvim_buf_get_lines(0, 0, -1, false))
end

Reading it
----------

1. "byteidx(charidx) ends at" is the byte after the cluster, which is the
   cluster's last byte counted from 1: 12 for the 11-byte ZWJ sequence that
   starts at byte 2; 9 for the skin-toned thumb and the flag (8 bytes each);
   6 for 漢 with its combining acute (5 bytes). str_utf_end ends at the first
   codepoint: byte 5 for the three emoji, byte 4 for 漢. For a plain 漢 both
   give 4.
2. A block whose left edge is column 2 cuts the double-width 👍🏽: both ends of
   line 2's region are byte 2, the cluster's first byte, the end with off 1.
2b. Only the edge that cuts the character is given as its first byte, with a
   non-zero off. A one-column block on it (gg0l<C-V>jj, section 2's case)
   gives both ends there. A block whose right edge is on its left half
   (gg0<C-V>ljj) starts at byte 1 and ends at byte 2, off 1. A block whose
   left edge is on its right half starts at byte 2, off 1, and ends at byte
   10 when it reaches past the character (gg0ll<C-V>ljj), or at byte 2, off 2,
   when it is one column wide (gg0ll<C-V>jj).
3. With 'selection' "old", gg0lvj"_d removes "bc" and no line break: the end
   on the empty line moved to the end of the line above. Started in the
   indent, gg0lvjj"_d removes lines 1 and 2 whole. Under "inclusive" the
   same keys join the lines.
4. `.` after a `:normal! "_d` of "beta" removed " gam": 4 characters at the
   cursor, the size of the first removal. After nvim_buf_set_text(), `.` repeated the `x`
   made before it: the API call did not set what `.` repeats.
5. getline() gives the NUL as "\n"; nvim_buf_get_lines() gives "\0". A Lua
   string holding a NUL is a Blob to Vimscript (type 10, v:t_blob): charidx()
   refuses it with E1174, byteidx() and strchars() with E976.
6. One typed key, <F2>, runs a mapping that empties the buffer and puts its
   line back. With no undo break between the two, the first u leaves "text"
   as it was and the second empties it: the removal and the put-back were one
   undo block. With `let &g:undolevels = &g:undolevels` between them, the
   first u empties the buffer and the second brings "text" back. Keys fed
   with nvim_feedkeys(..., 'mtx') in an `nvim -l` script make undo blocks of
   their own. The text before each "undo break" row is what Neovim printed
   while the keys ran: the mode and the undo messages.
