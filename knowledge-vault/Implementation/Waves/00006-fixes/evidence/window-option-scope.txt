# Whether a window's wrap, linebreak and breakindent reach other windows, measured by the orchestrator on 2026-09-26.
# Each probe runs as: nvim --clean -u NONE -i NONE --headless -l <probe>, first on the host's Homebrew Neovim, then on the downloaded 0.11.6 build. The global 'wrap' is off in both, as in a config that sets nowrap.

## Probe 1: the options set for the window (vim.wo[win])
#   local out = { 'nvim ' .. tostring(vim.version()) }
#   vim.o.wrap = false
#   local a = vim.api.nvim_get_current_win()
#   vim.wo[a].wrap = true
#   vim.wo[a].linebreak = true
#   vim.wo[a].breakindent = true
#   local function opts(w) return ('wrap=%s lbr=%s bri=%s'):format(tostring(vim.wo[w].wrap), tostring(vim.wo[w].linebreak), tostring(vim.wo[w].breakindent)) end
#   local file = vim.api.nvim_create_buf(true, false)
#   -- 1: split from the current window (A), win = -1
#   local w1 = vim.api.nvim_open_win(file, false, { split = 'left', win = -1 })
#   table.insert(out, 'split left, win=-1, current=A: ' .. opts(w1))
#   -- 2: split above A while the current window is another plain window
#   vim.api.nvim_set_current_win(w1)
#   vim.wo[w1].wrap = false; vim.wo[w1].linebreak = false; vim.wo[w1].breakindent = false
#   local w2 = vim.api.nvim_open_win(file, false, { split = 'above', win = a })
#   table.insert(out, 'split above, win=A, current=plain: ' .. opts(w2))
#   -- 3: :vsplit from A
#   vim.api.nvim_set_current_win(a)
#   vim.cmd('vsplit')
#   table.insert(out, ':vsplit from A: ' .. opts(vim.api.nvim_get_current_win()))
#   -- 4: a buffer entered into A later (wo = window value, applies to the new buffer)
#   local other = vim.api.nvim_create_buf(true, false)
#   vim.api.nvim_win_set_buf(a, other)
