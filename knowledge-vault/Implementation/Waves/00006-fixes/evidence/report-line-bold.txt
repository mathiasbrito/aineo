## The probe
-- What the screen shows for `[done]` under each way of making it bold: each
-- scenario runs in an editor whose own TUI runs in a pseudo-terminal
-- (truecolor), and the SGR the TUI writes before `[done]` on a full redraw
-- is decoded (fg as #rrggbb, B for bold). For contrast, `nvim__inspect_cell`
-- on the same cell. The time cell is checked the same way before `09:05`.

local nvim = vim.v.progpath

local function start()
  local address = vim.fn.tempname()
  local state = { out = '' }
  if os.getenv('PROBE_HEADLESS') == '1' then
    state.job = vim.fn.jobstart({ nvim, '--clean', '-n', '--listen', address, '--headless', '--cmd', 'set lines=8 columns=60' })
    vim.wait(5000, function()
      local ok, c = pcall(vim.fn.sockconnect, 'pipe', address, { rpc = true })
      state.ch = ok and c or nil
      return state.ch ~= nil
    end, 10)
    return state
  end
  state.job = vim.fn.jobstart({ nvim, '--clean', '-n', '--listen', address }, {
    pty = true,
    width = 60,
    height = 8,
    env = { TERM = 'xterm-256color', COLORTERM = 'truecolor' },
    on_stdout = function(_, d)
      local chunk = table.concat(d, '\n')
      state.out = state.out .. chunk
      local _, n = chunk:gsub('\27%[5n', '')
      for _ = 1, n do
        vim.fn.chansend(state.job, '\27[0n')
      end
    end,
  })
  vim.wait(5000, function()
    local ok, c = pcall(vim.fn.sockconnect, 'pipe', address, { rpc = true })
    state.ch = ok and c or nil
    return state.ch ~= nil
  end, 10)
  vim.wait(5000, function()
    return state.out:find('~', 1, true) ~= nil
  end, 10)
  return state
end

local SETUP = [[
  vim.o.termguicolors = true
  vim.api.nvim_buf_set_lines(0, 0, -1, false, { '09:05 [done] Task - Summary' })
  _G.ns = vim.api.nvim_create_namespace('probe')
  _G.mark = function(first, last, opts)
    opts.end_col = last
    return vim.api.nvim_buf_set_extmark(0, _G.ns, 0, first, opts)
  end
  _G.define = function()
    vim.cmd('highlight default link AineoReportTime Comment')
    vim.cmd('highlight default link AineoReportDone DiagnosticOk')
  end
  _G.bold_group = function()
    vim.cmd('highlight default link AineoReportStatusBold @markup.strong')
  end
  _G.marks = function(bold_opts)
    mark(0, 5, { hl_group = 'AineoReportTime' })
    mark(6, 12, { hl_group = 'AineoReportDone' })
    if bold_opts then
      bold_opts.hl_group = bold_opts.hl_group or 'AineoReportStatusBold'
      mark(6, 12, bold_opts)
    end
  end
]]

--- The SGR state in force before the last occurrence of `text` in `out`:
--- fg and bold, the whole stream up to it parsed.
local function sgr_before(out, text)
  local at, from = nil, 1
  while true do
    local found = out:find(text, from, true)
    if not found then break end
    at, from = found, found + 1
  end
  if not at then
    return 'not drawn'
  end
  local fg, bold = '-', false
  for params in out:sub(1, at - 1):gmatch('\27%[([%d;]*)m') do
    local list = vim.split(params, ';', { plain = true })
    local i = 1
    while i <= #list do
      local p = list[i]
      if p == '' or p == '0' then
        fg, bold = '-', false
      elseif p == '1' then
        bold = true
      elseif p == '22' then
        bold = false
      elseif p == '39' then
        fg = '-'
      elseif (p == '48' or p == '58') and list[i + 1] == '2' then
        i = i + 4
      elseif (p == '48' or p == '58') and list[i + 1] == '5' then
        i = i + 2
      elseif p == '38' and list[i + 1] == '2' then
        fg = ('#%02x%02x%02x'):format(tonumber(list[i + 2]), tonumber(list[i + 3]), tonumber(list[i + 4]))
        i = i + 4
      end
      i = i + 1
    end
  end
  return fg .. (bold and '+B' or '')
end

io.stdout:write(tostring(vim.version()) .. '\n')
io.stdout:write('screen (TUI SGR): time before "09:05", status before "[done]"; inspect_cell of "[" for contrast\n')
for _, s in ipairs(scenarios) do
  local only = os.getenv('PROBE_ONLY')
  if only and s[1]:sub(1, #only) ~= only then goto continue end
  do
  local st = start()
  vim.rpcrequest(st.ch, 'nvim_exec_lua', SETUP, {})
  local ok, err = pcall(vim.rpcrequest, st.ch, 'nvim_exec_lua', s[2], {})
  vim.wait(200)
  if os.getenv('PROBE_DUMP') == '1' then io.stdout:write('#before=' .. #st.out .. ' ' .. (st.out:sub(-400):gsub('\27', '<E>')) .. '\n') end
  vim.wait(600)
  local inspect = vim.rpcrequest(st.ch, 'nvim_exec_lua', [[
    local c = vim.api.nvim__inspect_cell(1, 0, 6)
    return (c[2].foreground and ('#%06x'):format(c[2].foreground) or '-') .. (c[2].bold and '+B' or '')
  ]], {})
  local info = vim.rpcrequest(st.ch, 'nvim_exec_lua', 'return _G.info', {})
  if os.getenv('PROBE_DUMP') == '1' then io.stdout:write('#out=' .. #st.out .. ' ' .. (st.out:sub(1, 600):gsub('\27', '<E>')) .. '\n') end
  io.stdout:write(('%-74s time=%-9s status=%-12s inspect_cell=%-10s%s%s\n'):format(
    s[1], sgr_before(st.out, '09:05'), sgr_before(st.out, '[done]'), inspect,
    ok and '' or (' SCENARIO ERROR: ' .. tostring(err)),
    (info ~= vim.NIL and info) and ('  ' .. info) or ''))
  pcall(vim.rpcrequest, st.ch, 'nvim_command', 'qa!')
  vim.fn.jobwait({ st.job }, 3000)
  vim.fn.jobstop(st.job)
  end
  ::continue::
end
