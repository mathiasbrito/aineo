-- Which mode a child Neovim is in right after keys that enter Insert or
-- Terminal mode, by how the keys were given: pending (nvim_input, as
-- mini.test's child.type_keys gives them) or executed at once
-- (nvim_feedkeys with 'x', as a helper's press() gives them).
-- Each case runs in a fresh child: nvim --clean --headless --embed.

local function child()
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.keymap.set('n', '<Space>i', function() vim.cmd.startinsert() end)
    vim.keymap.set('n', '<Space>m', function()
      vim.cmd.startinsert()
      _G.mode_inside = vim.api.nvim_get_mode().mode
    end)
    vim.keymap.set('n', '<Space>w', function()
      vim.cmd.startinsert()
      vim.cmd.wincmd('w')
    end)
  ]], {})
  return chan
end

local function terminal(chan)
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.cmd.enew()

end

==== script probe-highlight.lua ====
-- What `:highlight default link G Title` does to a group G that something
-- was given first, and which default link `:highlight clear` restores.
-- Each case runs in a fresh child (--clean --headless --embed); the "given
-- first" command reaches it typed on the command line (nvim_input), as an
-- API command (nvim_command) or from Lua (vim.cmd in nvim_exec_lua).

local function child()
  return vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
end

local function definition(chan)
  return vim.inspect(vim.rpcrequest(chan, 'nvim_get_hl', 0, { name = 'G' }), { newline = ' ', indent = '' })
end

local function given(chan, how, command)
  if how == 'typed' then
    vim.rpcrequest(chan, 'nvim_input', ':' .. command .. '<CR>')
    vim.wait(200)
  elseif how == 'api' then
    vim.rpcrequest(chan, 'nvim_command', command)
  else
    vim.rpcrequest(chan, 'nvim_exec_lua', 'vim.cmd(...)', { command })
  end
end

local out = { tostring(vim.version()) }
local cases = {
  { 'G gui=NONE cterm=NONE', 'lua' },
  { 'G gui=NONE cterm=NONE', 'api' },
  { 'G guifg=#ff0000', 'lua' },
  { 'link G NONE', 'lua' },
  { 'link G NONE', 'api' },
  { 'link G NONE', 'typed' },
  { 'default link G DiagnosticOk', 'lua' },
}
for _, case in ipairs(cases) do
  local chan = child()
  given(chan, case[2], 'highlight ' .. case[1])
  local before = definition(chan)
  vim.rpcrequest(chan, 'nvim_command', 'highlight default link G Title')
  local after = definition(chan)
  vim.rpcrequest(chan, 'nvim_command', 'highlight clear')
  local cleared = definition(chan)
  out[#out + 1] = ('given first (%s) :highlight %s -> %s; then default link G Title -> %s; then :highlight clear -> %s'):format(
    case[2],
    case[1],
    before,
    after,
    cleared
  )
  vim.fn.jobstop(chan)
end
io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-system-exit.lua ====
-- When vim.system() reports the exit of a process that leaves a descendant
-- holding its output: `sh -c 'sleep 3 & exit 3'` exits at once with code 3,
-- while its `sleep 3` inherits sh's stdout and stderr.
-- Compared with vim.uv.spawn, whose output goes to a file.

local here = vim.fn.fnamemodify(arg[0], ':p:h')
local COMMAND = { 'sh', '-c', 'sleep 3 & exit 3' }
local out = { tostring(vim.version()) }
local function log(s)
  out[#out + 1] = s
end
local function ms_since(t0)
  return ('%.0f'):format((vim.uv.hrtime() - t0) / 1e6)
end

-- 1. vim.system with its output captured (the default).
local t0 = vim.uv.hrtime()
local result
vim.system(COMMAND, {}, function(r)
  result = r
  result.at = ms_since(t0)
end)
vim.wait(10000, function()
  return result ~= nil
end, 10)
log(('vim.system, output captured: on_exit after %s ms, code=%s'):format(result.at, result.code))

-- 2. vim.system with stdout and stderr not captured.
t0 = vim.uv.hrtime()
result = nil
vim.system(COMMAND, { stdout = false, stderr = false }, function(r)
  result = r
  result.at = ms_since(t0)
end)
vim.wait(10000, function()
  return result ~= nil
end, 10)
log(('vim.system, stdout = false, stderr = false: on_exit after %s ms, code=%s'):format(result.at, result.code))

-- 3. SystemObj:wait(500) on the captured form.
t0 = vim.uv.hrtime()
local waited = vim.system(COMMAND):wait(500)
log(('vim.system(...):wait(500): returned after %s ms, result=%s'):format(ms_since(t0), waited and ('code=' .. waited.code) or 'nil'))
vim.wait(4000)

-- 4. vim.uv.spawn, stdout and stderr to a file.
local path = here .. '/system-exit-spawn-output.txt'
local fd = assert(vim.uv.fs_open(path, 'w', 420))
t0 = vim.uv.hrtime()
local spawned
local handle = vim.uv.spawn('sh', { args = { '-c', 'sleep 3 & exit 3' }, stdio = { nil, fd, fd } }, function(code, signal)
  spawned = { at = ms_since(t0), code = code, signal = signal }
end)
vim.uv.fs_close(fd)
vim.wait(10000, function()
  return spawned ~= nil
end, 10)
handle:close()
log(('vim.uv.spawn, output to a file: exit callback after %s ms, code=%s'):format(spawned.at, spawned.code))
vim.wait(3500)

io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-jobs.lua ====
-- 1. jobwait({id}, 0) and vim.uv.kill(pid, 0) on a job jobstop() has stopped
--    but that still runs: it ignores SIGHUP and SIGTERM.
-- 2. The order of TermClose and the job's on_exit when a terminal job ends.

local DEAF = { 'sh', '-c', 'trap "" HUP TERM; while :; do sleep 1; done' }
local out = { tostring(vim.version()) }
local function log(s)
  out[#out + 1] = s
end
local function ms_since(t0)
  return ('%.0f'):format((vim.uv.hrtime() - t0) / 1e6)
end

local function stopped_job(label, start)
  local id = start()
  local pid = vim.fn.jobpid(id)
  vim.wait(300)
  vim.fn.jobstop(id)
  local t0 = vim.uv.hrtime()
  local alive = vim.uv.kill(pid, 0)
  log(('%s: vim.uv.kill(pid, 0) right after jobstop() returned %s in %s ms'):format(label, tostring(alive), ms_since(t0)))
  t0 = vim.uv.hrtime()
  local status = vim.fn.jobwait({ id }, 0)[1]
  log(('%s: jobwait({id}, 0) returned %d after %s ms'):format(label, status, ms_since(t0)))
  local gone, err = vim.uv.kill(pid, 0)
  log(('%s: vim.uv.kill(pid, 0) afterwards returned %s %s'):format(label, tostring(gone), tostring(err)))
end

stopped_job('pipe job', function()
  return vim.fn.jobstart(DEAF)
end)
stopped_job('terminal job', function()
  vim.cmd.enew()
  return vim.fn.jobstart(DEAF, { term = true })
end)

for run = 1, 5 do
  local order = {}
  vim.cmd.enew()
  local buf = vim.api.nvim_get_current_buf()
  vim.api.nvim_create_autocmd('TermClose', {
    buffer = buf,
    once = true,

==== script probe-bwipeout.lua ====
-- When :bwipeout! of a terminal buffer shown in the current window puts
-- another buffer in that window: before BufWipeout or after it, for a
-- terminal whose job still runs and for one whose job has ended. The other
-- window shows an unlisted scratch buffer, so no listed buffer is left.

local out = { tostring(vim.version()) }

local function trace(label, job_ends_first)
  vim.cmd('silent! %bwipeout!')
  vim.cmd.enew()
  local scratch = vim.api.nvim_get_current_buf()
  vim.bo[scratch].buflisted = false
  vim.bo[scratch].buftype = 'nofile'
  vim.cmd.vsplit()
  vim.cmd.enew()
  local win = vim.api.nvim_get_current_win()
  local job = vim.fn.jobstart(job_ends_first and { 'sh', '-c', 'exit 0' } or { 'cat' }, { term = true })
  local term = vim.api.nvim_get_current_buf()
  vim.bo[term].buflisted = true
  if job_ends_first then
    vim.fn.jobwait({ job }, 2000)
    vim.wait(200)
  end
  local events = {}
  local group = vim.api.nvim_create_augroup('probe_bwipeout', {})
  for _, event in ipairs({ 'TermClose', 'BufUnload', 'BufWipeout', 'BufWinEnter', 'BufEnter' }) do
    vim.api.nvim_create_autocmd(event, {
      group = group,
      callback = function(args)
        local shown = vim.api.nvim_win_is_valid(win) and vim.api.nvim_win_get_buf(win)
        events[#events + 1] = ('%s(buf %s; the window shows %s)'):format(
          event,
          args.buf == term and 'term' or tostring(args.buf),
          shown == term and 'term' or (shown and ('buf ' .. shown) or 'nothing: closed')
        )
      end,
    })
  end
  vim.cmd('bwipeout!')
  vim.api.nvim_del_augroup_by_id(group)
  out[#out + 1] = ('%s: %s'):format(label, table.concat(events, ' -> '))
  out[#out + 1] = ('%s: afterwards the window %s'):format(
    label,
    vim.api.nvim_win_is_valid(win)
        and ('shows buf %d, name %q, listed %s'):format(
          vim.api.nvim_win_get_buf(win),
          vim.api.nvim_buf_get_name(vim.api.nvim_win_get_buf(win)),
          tostring(vim.bo[vim.api.nvim_win_get_buf(win)].buflisted)
        )
      or 'is closed'
  )
  vim.cmd('silent! only')
end

trace('running terminal', false)
trace('ended terminal', true)
io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-kitty-ctrl-c.lua ====
-- What a program in a Neovim terminal receives when Ctrl-C is typed in
-- Terminal mode, without the kitty keyboard protocol and with it enabled by
-- the program (CSI > 1 u, CSI > 5 u). The key is typed into a child Neovim
-- (--clean --headless --embed) with nvim_input(), as mini.test's
-- child.type_keys() types keys.

local here = vim.fn.fnamemodify(arg[0], ':p:h')
local out = { tostring(vim.version()) }

for _, flags in ipairs({ 'none', '1', '5' }) do
  local received_path = ('%s/kitty-received-%s-%s.txt'):format(here, flags, tostring(vim.version()))
  vim.fn.delete(received_path)
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.cmd.enew()
    vim.fn.jobstart({ '/usr/bin/python3', ... }, { term = true })
  ]], { here .. '/keys-reader.py', received_path, flags })
  vim.wait(3000, function()
    local lines = vim.rpcrequest(chan, 'nvim_buf_get_lines', 0, 0, -1, false)
    return table.concat(lines, '\n'):find('ready', 1, true) ~= nil
  end, 20)
  vim.rpcrequest(chan, 'nvim_command', 'startinsert')
  vim.wait(200)
  local mode = vim.rpcrequest(chan, 'nvim_get_mode').mode
  vim.rpcrequest(chan, 'nvim_input', '<C-c>')
  vim.wait(5000, function()
    return vim.uv.fs_stat(received_path) ~= nil
  end, 50)
  vim.wait(100)
  local received = table.concat(vim.fn.readfile(received_path), '\n')
  out[#out + 1] = ('flags %-4s mode before the key: %s; the program received %s'):format(flags, mode, received)
  vim.fn.jobstop(chan)
end

io.stdout:write(table.concat(out, '\n') .. '\n')

==== script keys-reader.py (the program in the terminal) ====
"""Runs inside a Neovim terminal: optionally enables the kitty keyboard
protocol with the given flags (CSI > flags u), asks for the flags in force
(CSI ? u), then records every byte its terminal sends for a few seconds.
Usage: keys-reader.py <out file> <flags or 'none'>"""
import os
import select
import sys
import termios
import time
import tty

out_path, flags = sys.argv[1], sys.argv[2]
fd = sys.stdin.fileno()
tty.setraw(fd)
if flags != 'none':
    os.write(sys.stdout.fileno(), ('\x1b[>%su' % flags).encode())
    os.write(sys.stdout.fileno(), b'\x1b[?u')
os.write(sys.stdout.fileno(), b'ready\r\n')
received = b''
deadline = time.time() + 4
while time.time() < deadline:
    ready, _, _ = select.select([fd], [], [], 0.1)
    if ready:
        received += os.read(fd, 1024)
with open(out_path, 'w') as out:
    out.write(repr(received))
