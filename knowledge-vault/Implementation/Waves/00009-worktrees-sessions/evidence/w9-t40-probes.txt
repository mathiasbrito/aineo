================================================================================

Run by the T40 planning agent (Claude, Opus 5.5) on 2026-10-07, Neovim 0.12.5 (Homebrew),
macOS arm64, on `dev` 85a57f9, with T35's branch read at c2a6cee (PR #137) and T36's at
fe112c4 (PR #135). P1–P3 ran headless in a Neovim isolated by the runner below: every XDG
directory but the runtime one, the log and CLAUDE_CONFIG_DIR under a scratch directory inside
the agent's worktree, `--clean`, no shada. TMPDIR is kept, since `stdpath('run')` falls back

written. P4 is a read-only look at the host's process table (`ps`), no file read.

session id is cut to its first 8 hex digits, as `w9-real-claude-sessions.txt` writes them.

Findings
--------

P1  `stdpath('run')` on macOS, with XDG_RUNTIME_DIR unset, is a directory of each process's
    own, `<TMPDIR>/nvim.<user>/<6 random characters>`, and `v:servername` is
    `<that directory>/nvim.<pid>.0`; two Neovims never share it, and a Neovim that quits
    removes it. So a registry of running editors cannot live there: `stdpath('state')` is the
    per-user place two Neovims share. A relay-shaped child (`--clean --cmd 'set
    noloadplugins'`, the environment inherited) has the same `stdpath('state')` as its
    parent. With XDG_RUNTIME_DIR set, `stdpath('run')` is that directory itself, shared;
    the probe's own was too long for a socket (115 bytes against macOS's 104), so no server
    started there — W-3's limit again, not a finding about Neovim.
P2  A hook command run through a shell — `/bin/sh -c`, `/bin/zsh -c`, `/bin/bash -c`, and
    Node's `spawn(command, { shell: true })` — expands `"$PPID"` to the pid of the process
    that ran the shell (the stand-in for Claude Code) in every case. The relay's own parent
    (`vim.uv.os_getppid()`) is that process only when the shell runs the command in its own
    place: one command under any of the three shells and Node; with two commands
    (`true; <command>`), `/bin/sh` and `/bin/bash` and Node's shell leave the shell between
    them, so the relay's parent is the shell. So a relay that must name Claude Code's
    process takes it from `"$PPID"` written on its command line, never from its own parent.
    How Claude Code itself runs a command hook was not measured.
P3  A connection to an editor that is gone fails at once: ENOENT after the editor quit (its
    socket file is gone; 0.1 ms) — the incident's error — and ECONNREFUSED after it was
    killed by SIGKILL (its socket file and runtime directory stay; 0.0 ms). A registry entry
    of either kind costs a reader nothing to skip.
P4  The incident, in the host's process table (2026-10-07, about 14:36): the orchestrator's
    Claude Code session runs under Claude Code's background host — `claude --bg-pty-host
    /tmp/cc-daemon-<uid>/<id>/pty/55110cd0….sock … -- <home>/.local/share/claude/versions/2.1.291
    --session-id 55110cd0… --fork-session --resume <home>/.claude/projects/<folder>/938616f1….jsonl
    --mcp-config {…} --allowed-tools …`, parent pid 1, started 2026-10-06 16:20:18. So
    the conversation moved to the background is a new Claude Code process on a new session
    id, 55110cd0…, forked from the transcript of 938616f1…. Its `--mcp-config` is aineo's,
    carried over word for word, `AINEO_EDITOR_ADDRESS` naming
    `<TMPDIR>/nvim.<user>/Zsm2hJ/nvim.64474.0`, an editor (pid 64474) that no longer runs.
    The background host's command line carries `--mcp-config` and the allowed tools
    (respelled `--allowed-tools`) but no `--append-system-prompt`; whether it would carry a
    `--settings` (T35's hooks) cannot be read from it, since this session was started
    before T35. Its MCP server, aineo's relay (`nvim --headless --clean --cmd set
    noloadplugins -l <checkout>/lua/aineo/mcp/relay.lua`), has Claude Code's process
    (2.1.291) as its parent, as M5 measured, and in its environment `NVIM` and
    `AINEO_CHILD=1` — Claude Code passes its environment on to its MCP server — and
    `CLAUDE_CODE_SESSION_ID=55110cd0…`, the process's own first session. Two other
    aineo-started Claude Codes on the host, each in a live Neovim (its parent), show the
    ordinary shape: `claude --session-id …` or `--resume …`, the relay a child of claude.

Not measured: how Claude Code runs a command hook (through which shell, and whether its

-- P1: where a Neovim's runtime directory and server address are, on macOS, and
-- whether two Neovims share the runtime directory; whether a relay started as
-- aineo starts its MCP server (`--headless --clean --cmd 'set noloadplugins'`)
-- sees the editor's stdpath('state').
local here = os.getenv('PROBE_HERE')
local function show(...)
  io.stdout:write(table.concat(vim.tbl_map(function(v)
    return type(v) == 'string' and v or vim.inspect(v, { newline = ' ', indent = '' })
  end, { ... }), '  '), '\n')
end
local function redact(text)
  text = text:gsub(vim.pesc(vim.env.TMPDIR or '/nonexistent'), '<TMPDIR>/')
  text = text:gsub(vim.pesc(here), '<probe>')
  text = text:gsub('nvim%.[%w_]+/', 'nvim.<user>/')
  return text
end
local QUERY = 'io.stdout:write(vim.fn.stdpath("run") .. "|" .. vim.v.servername .. "|" .. vim.fn.stdpath("state") .. "|" .. vim.uv.os_getpid())'
local function child(extra_env, args)
  local command = { vim.v.progpath, '--headless', '--clean', '-i', 'NONE' }
  vim.list_extend(command, args or {})
  vim.list_extend(command, { '--cmd', 'lua ' .. QUERY, '+qa!' })
  local result = vim.system(command, { env = extra_env, text = true }):wait()
  return result.stdout
end
show('this Neovim (the runner):  run =', redact(vim.fn.stdpath('run')), ' servername =', redact(vim.v.servername), ' XDG_RUNTIME_DIR =', tostring(vim.env.XDG_RUNTIME_DIR))
for i = 1, 2 do
  local run, server, state, pid = unpack(vim.split(child({ NVIM = '' }), '|', { plain = true }))
  show(('child %d (pid %s):'):format(i, pid), ' run =', redact(run), ' servername =', redact(server), ' state =', redact(state))

// P2's Node stand-in for Claude Code: runs one hook command string as Node's
// `child_process.spawn(command, { shell: true })` runs it (`/bin/sh -c` on
// macOS), and prints its own pid first. Not Claude Code: how Claude Code

const { spawn } = require('child_process');
const command = process.argv[2];
process.stdout.write(`node_pid=${process.pid}\n`);

child.stdin.end('{"session_id":"x"}');
child.on('exit', (code) => process.stdout.write(`hook_exit=${code}\n`));

-- P2: which process a hook relay's `$PPID` and its own parent name, when a
-- command hook is run through a shell, as Claude Code runs one (D-docs; T35's
-- brief: "Claude Code runs it through a shell"). Stand-ins for Claude Code:
-- this Neovim running `/bin/sh -c` and `/bin/zsh -c` (vim.system), and Node's
-- `spawn(command, { shell: true })`. The real `claude` never runs.
local here = os.getenv('PROBE_HERE')
local node = os.getenv('PROBE_NODE')
local function show(...)
  io.stdout:write(table.concat(vim.tbl_map(function(v)
    return type(v) == 'string' and v or vim.inspect(v, { newline = ' ', indent = '' })
  end, { ... }), '  '), '\n')
end
local function quoted(word)
  return "'" .. word:gsub("'", [['\'']]) .. "'"
end
local record = here .. '/p2-record.lua'
local function hook_command(log, prefix)
  local words = vim.tbl_map(quoted, { vim.v.progpath, '--headless', '--clean', '--cmd', 'set noloadplugins', '-l', record, log })
  return (prefix or '') .. table.concat(words, ' ') .. ' "$PPID"'
end
local function read_log(log)
  local lines = vim.fn.readfile(log)
  vim.fn.delete(log)
  return lines[1] or '(nothing written)'
end
local me = vim.uv.os_getpid()
for _, shell in ipairs({ '/bin/sh', '/bin/zsh', '/bin/bash' }) do
  for _, shape in ipairs({ { 'one command', nil }, { 'two commands', 'true; ' } }) do
    local log = here .. '/p2.log'
    local result = vim.system({ shell, '-c', hook_command(log, shape[2]) }, { stdin = '{"session_id":"x"}' }):wait()
    show(('%s -c, %s: the stand-in Claude pid=%d;'):format(shell, shape[1], me), read_log(log), 'exit', result.code)
  end
end
if node then
  for _, shape in ipairs({ { 'one command', nil }, { 'two commands', 'true; ' } }) do
    local log = here .. '/p2.log'
    local result = vim.system({ node, here .. '/p2-node-claude.js', hook_command(log, shape[2]) }, { text = true }):wait()
    show(("node spawn(shell: true), %s: %s"):format(shape[1], (result.stdout:gsub('\n', ' '))), read_log(log))
  end
end

P3: a connection to an editor that is gone
------------------------------------------

-- P3: what a connection to an editor's address gives once that editor is gone,
-- through the MCP relay's own connect (libuv's pipe connect, as
-- `lua/aineo/mcp/editor.lua` makes it): after the editor quit, and after it
-- was killed by SIGKILL; and how long the failure takes.
local here = os.getenv('PROBE_HERE')
local function show(...)
  io.stdout:write(table.concat(vim.tbl_map(function(v)
    return type(v) == 'string' and v or vim.inspect(v, { newline = ' ', indent = '' })
  end, { ... }), '  '), '\n')
end
local function start_editor()
  local file = here .. '/p3-address'
  vim.fn.delete(file)
  local editor = vim.system({ vim.v.progpath, '--headless', '--clean', '-i', 'NONE', '--cmd',
    ('lua vim.fn.writefile({ vim.v.servername }, %q)'):format(file) })
  vim.wait(3000, function() return vim.uv.fs_stat(file) ~= nil end, 10)
  return editor, vim.fn.readfile(file)[1]
end
local function try_connect(address)
  local pipe = assert(vim.uv.new_pipe(false))
  local result
  local start = vim.uv.hrtime()
  pipe:connect(address, function(failure)
    result = failure or 'connected'
  end)
  vim.wait(2000, function() return result ~= nil end, 1)
  pipe:close()
  return result, (vim.uv.hrtime() - start) / 1e6
end
local editor, address = start_editor()
show('live editor:', try_connect(address))
local channel = vim.fn.sockconnect('pipe', address, { rpc = true })
pcall(vim.rpcrequest, channel, 'nvim_command', 'qa!')
editor:wait(3000)
local result, ms = try_connect(address)
show('after the editor quit:', result, ('%.1f ms'):format(ms), 'socket file left:', vim.uv.fs_stat(address) ~= nil)
editor, address = start_editor()
editor:kill(9)
editor:wait(3000)
result, ms = try_connect(address)
show('after SIGKILL:', result, ('%.1f ms'):format(ms), 'socket file left:', vim.uv.fs_stat(address) ~= nil, 'its runtime directory left:', vim.uv.fs_stat(vim.fs.dirname(address)) ~= nil)
vim.fn.delete(vim.fs.dirname(address), 'rf')
