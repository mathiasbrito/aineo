then a start that resumes S, which finds no conversation, so the claude
home starts a new id (R) in its place. Every callback noted with the
milliseconds since the first start, a counter of main-loop ticks, what the
claude home says inside it (`session_id()`, `session_status()`), and
whether it ran in a fast event.

  `session_status()` is `starting`.
- `on_session_switched` gets id = R, source = `startup`, left = S (the id
  that found no conversation), reason = nil.
- R is ready 1547 ms after the fallback (4571 ms after the first start).

P1b — `on_terminal_replaced` outside the fallback
-------------------------------------------------

-- T41 planning probes, run with `make test_file FILE=<this file>` on dev 2fdda81.
-- Each case prints what it measured as `PROBE <name>: <json>` on stderr.
local MiniTest = require('mini.test')
local children = dofile('tests/helpers/child.lua')
local claude = dofile('tests/helpers/claude_session.lua')
local fixture = dofile('tests/helpers/fixture.lua')
local git_repo = dofile('tests/helpers/git_repo.lua')

local child = MiniTest.new_child_neovim()

local T = MiniTest.new_set({
  hooks = {
    pre_case = function()
      children.restart(child)
    end,
    post_once = child.stop,
  },
})

local function show(name, value)
  io.stderr:write(('PROBE %s: %s\n'):format(name, vim.json.encode(value)))
end

-- P1. Which callbacks T19's fallback makes, in which order, in one tick or not, and what the
-- claude home says inside each; then when the replacement is ready.
local START_NOTING_ALL = [[
  local helper, overrides = dofile(...), select(2, ...)
  _G.events = _G.events or {}
  _G.tick = _G.tick or 0
  if not _G.ticking then
    _G.ticking = true
    local timer = vim.uv.new_timer()
    timer:start(0, 1, vim.schedule_wrap(function() _G.tick = _G.tick + 1 end))
  end
  local function note(kind, extra)
    local claude = require('aineo.claude')
    table.insert(_G.events, vim.tbl_extend('force', {
      kind = kind,
      ms = math.floor(vim.uv.hrtime() / 1e6),
      tick = _G.tick,
      session_id = claude.session_id(),
      status = claude.session_status(),
      fast = vim.in_fast_event(),
    }, extra or {}))
  end
  local settings = helper.stand_in_settings(overrides)
  settings.on_terminal_replaced = function(terminal)
    note('on_terminal_replaced', { terminal = terminal })
  end
  settings.on_session_switched = function(id, source, left, reason)
    note('on_session_switched', { id = id, source = source, left = left, reason = reason or vim.NIL })
  end
  note('start_session called')
  local buffer = require('aineo.claude').start_session(settings)
  note('start_session returned')
  vim.api.nvim_win_set_buf(0, buffer)
  return buffer
]]

local function start_noting_all(fake, overrides)
  child.lua('for name, value in pairs(...) do vim.env[name] = value end', { fake.environment })
  return child.lua(START_NOTING_ALL, { vim.fs.joinpath(vim.fn.getcwd(), 'tests', 'helpers', 'claude_session.lua'), overrides })
end

local function poll_ready(label)
  vim.wait(claude.PATIENCE_MS, function()
    return child.lua_get("require('aineo.claude').session_status()") == 'ready'
  end, 10)
  child.lua(
    [[table.insert(_G.events, { kind = ..., ms = math.floor(vim.uv.hrtime() / 1e6), status = require('aineo.claude').session_status(), session_id = require('aineo.claude').session_id() })]],
    { label }
  )
end

T['P1 fallback callbacks'] = function()
  local fake = claude.fake('t41-p1', 'ready', {
    AINEO_FAKE_CLAUDE_CONVERSATIONS = fixture.directory('t41-p1-conversations'),
  })
  local state = { state_directory = fixture.directory('t41-p1-state') }
  local first = start_noting_all(fake, state)
  claude.wait_for_start(fake)
  poll_ready('first ready')
  child.lua('vim.fn.jobstop(vim.bo[...].channel)', { first })
  claude.wait_for_status(child, 'exited')
  start_noting_all(fake, state)
  claude.wait_for_starts(fake, 3)
  vim.wait(claude.PATIENCE_MS, function()
    return #vim.tbl_filter(function(event)
      return event.kind == 'on_session_switched'
    end, child.lua_get('_G.events')) >= 1
  end, 10)
  poll_ready('replacement ready')
  local events = child.lua_get('_G.events')
  local base = events[1].ms
  for _, event in ipairs(events) do
    event.ms = event.ms - base
  end
  show('P1 events', events)
  show('P1 ids', {
    first = claude.words_after(claude.start_arguments(fake, 1), '--session-id')[1],
    second_resumes = claude.words_after(claude.start_arguments(fake, 2), '--resume')[1],
    third = claude.words_after(claude.start_arguments(fake, 3), '--session-id')[1],
  })
end

-- P2. The strand at the homes, on dev: what a follow of the replacement does to the dead
-- session's draft and records when the home followed it.
local X = '11111111-1111-4111-8111-111111111111'
local Y = '22222222-2222-4222-8222-222222222222'

T['P2 draft and report strand'] = function()
  local state = fixture.directory('t41-p2-state')
  local result = child.lua(
    [[
    local state, X, Y = ...
    local draft = require('aineo.draft')
    local report = require('aineo.report')
    draft.set_draft_environment({ state_directory = state, working_directory = '/t41/p2' })
    report.set_report_environment({ clock = function() return '2026-10-08T10:00:00' end, state_directory = state, working_directory = '/t41/p2' })
    local input = vim.api.nvim_create_buf(false, true)
    vim.api.nvim_win_set_buf(0, input)
    draft.keep_draft(input)
    draft.follow_draft_session(X)
    report.follow_report_session(X)
    local shown = report.report_buffer()
    vim.api.nvim_buf_set_lines(input, 0, -1, false, { 'notes for the next prompt' })
    report.receive_report({ task = 'probe', status = 'done', summary = 'kept under X' })
    local function file(kind, id)
      return vim.fs.joinpath(state, 'aineo', kind, 'session-' .. vim.fn.sha256(id) .. (kind == 'drafts' and '.txt' or '.jsonl'))
    end
    local function read(path)
      local f = io.open(path, 'rb')
      if not f then return vim.NIL end
      local text = f:read('*a')
      f:close()
      return text
    end
    draft.follow_draft_session(Y)
    report.follow_report_session(Y)
    return {
      input_after_follow_of_Y = vim.api.nvim_buf_get_lines(input, 0, -1, false),
      x_draft = read(file('drafts', X)),
      y_draft = read(file('drafts', Y)),
      report_lines_after = vim.api.nvim_buf_get_lines(shown, 0, -1, false),
      x_records_lines = read(file('reports', X)) ~= vim.NIL and #vim.split(vim.trim(read(file('reports', X))), '\n') or 0,
      y_records = read(file('reports', Y)),
    }
    ]],
    { state, X, Y }
  )
  show('P2 draft and report', result)
end

T['P2b changes strand'] = function()
  git_repo.start_editor(child)
  local top = git_repo.create('t41-p2b', { ['a.txt'] = { 'a' } })
  local state = fixture.directory('t41-p2b-state')
  child.lua(
    [[
    local top = ...
    local changes = require('aineo.changes')
    changes.begin_session({ directory = top, show_diff = function() end })
    local pane = changes.pane_buffers()
    vim.api.nvim_win_set_buf(0, pane.files)
    ]],
    { top }
  )
  vim.wait(5000, function()
    return child.lua_get([[vim.api.nvim_buf_get_lines(vim.fn.bufnr('aineo://changes-files'), 0, 1, false)[1] or '']]) ~= ''
  end, 20)
  child.lua("require('aineo.changes').follow_changes_session({ id = ..., state_directory = select(2, ...) })", { X, state })
  vim.wait(500)
  git_repo.write(top, 'a.txt', { 'a', 'changed' })
  git_repo.commit_all(top, 'second')
  child.lua("vim.cmd.edit(vim.fn.fnameescape(...)); vim.cmd.write()", { vim.fs.joinpath(top, 'a.txt') })
  vim.wait(800)
  local function kept(id)
    local path = vim.fs.joinpath(state, 'aineo', 'changes-sessions', vim.fn.sha256(id) .. '.json')
    local f = io.open(path, 'rb')
    if not f then return vim.NIL end
    local text = f:read('*a')
    f:close()
    return vim.json.decode(text)
  end
  local x_before = kept(X)
  child.lua("require('aineo.changes').follow_changes_session({ id = ..., state_directory = select(2, ...) })", { Y, state })
  vim.wait(1500)
  local head = git_repo.git(top, { 'rev-parse', 'HEAD' })
  local x_after, y_after = kept(X), kept(Y)
  show('P2b changes', {
    x_kept_before_follow_of_Y = x_before and { base_is_first_commit = x_before.base ~= head, saved = x_before.saved } or vim.NIL,
    x_kept_after = x_after and { saved = x_after.saved } or vim.NIL,
    y_kept_after = y_after and { base_is_head_now = y_after.base == head, saved = y_after.saved } or vim.NIL,
  })
end

-- P3. A move of a session's file to another session's, by T36's records.move_records() and by a
-- plain link of a kept base: to missing, to present, from unreadable, from a symbolic link.
T['P3 moves'] = function()
  local dir = fixture.directory('t41-p3')
  local result = child.lua(
    [[
    local dir = ...
    local records = require('aineo.report.records')
    local function write(path, text, mode)
      local f = assert(io.open(path, 'wb')); f:write(text); f:close()
      if mode then vim.uv.fs_chmod(path, mode) end
    end
    local function state(path)
      local s = vim.uv.fs_lstat(path)
      return s and { type = s.type, mode = ('%o'):format(s.mode % 4096), nlink = s.nlink } or vim.NIL
    end
    local out = {}
    -- a: to missing
    write(dir .. '/a-from', 'x\n')
    out.a = { failure = records.move_records(dir .. '/a-from', dir .. '/a-to') or vim.NIL, from = state(dir .. '/a-from'), to = state(dir .. '/a-to') }
    -- b: to present
    write(dir .. '/b-from', 'x\n'); write(dir .. '/b-to', 'y\n')
    out.b = { failure = records.move_records(dir .. '/b-from', dir .. '/b-to') or vim.NIL, from = state(dir .. '/b-from'), to = state(dir .. '/b-to') }
    -- c: from unreadable (mode 000)
    write(dir .. '/c-from', 'x\n', tonumber('000', 8))
    out.c = { failure = records.move_records(dir .. '/c-from', dir .. '/c-to') or vim.NIL, from = state(dir .. '/c-from'), to = state(dir .. '/c-to'), readable = vim.uv.fs_access(dir .. '/c-to', 'R') }
    vim.uv.fs_chmod(dir .. '/c-to', tonumber('600', 8))
    -- d: from a symbolic link
    write(dir .. '/d-target', 'x\n'); vim.uv.fs_symlink(dir .. '/d-target', dir .. '/d-from')
    out.d = { failure = records.move_records(dir .. '/d-from', dir .. '/d-to') or vim.NIL, from = state(dir .. '/d-from'), to = state(dir .. '/d-to') }
    -- e: neither exists
    out.e = { failure = records.move_records(dir .. '/e-from', dir .. '/e-to') or vim.NIL, to = state(dir .. '/e-to') }
    -- f: a kept base linked to another session's name is read back whole
    local kept = require('aineo.changes.kept')
    local X, Y = '11111111-1111-4111-8111-111111111111', '22222222-2222-4222-8222-222222222222'
    kept.keep_base(dir, X, { top = '/repo', base = 'abc', saved = { 'a.txt' } })
    local linked = vim.uv.fs_link(kept.kept_base_file(dir, X), kept.kept_base_file(dir, Y))
    vim.uv.fs_unlink(kept.kept_base_file(dir, X))
    local y, y_unreadable = kept.read_kept_base(dir, Y)
    out.f = { linked = linked or vim.NIL, y = y or vim.NIL, y_unreadable = y_unreadable, x = state(kept.kept_base_file(dir, X)), y_mode = state(kept.kept_base_file(dir, Y)) }
    return out
    ]],
    { dir }
  )
  show('P3 moves', result)
end

return T
