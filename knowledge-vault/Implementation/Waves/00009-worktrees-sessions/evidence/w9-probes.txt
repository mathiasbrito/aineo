#!/bin/sh
# Runs a probe in a Neovim isolated under this directory: every XDG path, the log and
# CLAUDE_CONFIG_DIR here, no user config, no shada; git hermetic as the suites' own
# (tests/helpers/git.lua: no global or system configuration, a fixed author) with a
# ceiling at this directory, so no git run here ever finds the checkout's repository.
here=$(cd "$(dirname "$0")" && pwd)
home="$here/home"
mkdir -p "$home"
exec env -i PATH="/opt/homebrew/bin:/usr/bin:/bin" HOME="$home" TERM=xterm-256color \
  XDG_CONFIG_HOME="$home/config" XDG_DATA_HOME="$home/data" XDG_STATE_HOME="$home/state" \
  XDG_CACHE_HOME="$home/cache" NVIM_LOG_FILE="$home/nvim.log" CLAUDE_CONFIG_DIR="$home/claude" \
  GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 GIT_CEILING_DIRECTORIES="$here" \
  GIT_AUTHOR_NAME='aineo probe' GIT_AUTHOR_EMAIL=probe@aineo.invalid \
  GIT_COMMITTER_NAME='aineo probe' GIT_COMMITTER_EMAIL=probe@aineo.invalid \
  PROBE_HERE="$here" \
  nvim --clean --headless -i NONE -l "$@"

A3 (feature A): aineo's own watch (T23) as worktrees come, work and go; N watches; the cost of one read
----------------------------------------------------------------------------------------------------

-- A3: what aineo's own watch (aineo.git.watch_repository, T23, at this checkout's code)
-- calls back when worktrees come, work and go; what a plain fs_event on the common git
-- directory names; and what N watches cost when one worktree commits.
local here = os.getenv('PROBE_HERE')
local checkout = vim.fn.fnamemodify(here, ':h:h')
vim.opt.rtp:prepend(checkout)
local git_home = require('aineo.git')
local root = here .. '/a3'
local function show(...)
  io.stdout:write(table.concat(vim.tbl_map(function(v)
    return type(v) == 'string' and v or vim.inspect(v, { newline = ' ', indent = '' })
  end, { ... }), '  '), '\n')
end
local function git(cwd, ...)
  local r = vim.system(vim.list_extend({ 'git' }, { ... }), { cwd = cwd, text = true }):wait()
  return vim.trim(r.stdout .. r.stderr), r.code
end
vim.fn.delete(root, 'rf')
vim.fn.mkdir(root .. '/main', 'p')
local main = root .. '/main'
git(main, 'init', '-q', '-b', 'dev')
vim.fn.writefile({ '.claude/worktrees/' }, main .. '/.gitignore')
vim.fn.writefile({ 'a' }, main .. '/a.txt')
git(main, 'add', '.')
git(main, 'commit', '-qm', 'base')
local function find(directory)
  local found
  git_home.find_repository(directory, function(failure, repository)
    found = failure or repository
  end)
  vim.wait(5000, function() return found ~= nil end, 10)
  return found
end
local function watch(label, repository)
  local w, failure = git_home.watch_repository(repository, function(f, change)
    table.insert(CALLS, { label = label, at = math.floor((vim.uv.hrtime() - T0) / 1e6), failure = f and f.reason, change = change })
  end)
  assert(w, vim.inspect(failure))
  return w
end
T0 = vim.uv.hrtime()
