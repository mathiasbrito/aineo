local function row_starts(lnum)
  local text = lines[lnum]
  local starts, last_row = {}, nil
  for col = 1, #text do
    local pos = vim.fn.screenpos(win, lnum, col)
    if pos.row ~= 0 and pos.row ~= last_row then
      table.insert(starts, { row = pos.row, screen_col = pos.col, char = vim.fn.strpart(text, col - 1, 1, true) })
      last_row = pos.row
    end
  end
  return starts
