==== script rm-pairs-order.lua (written by PR #97's re-measure, unchanged) ====
-- Prints the order pairs() visits 12 string keys in, to see whether it
-- changes from one Neovim process to the next.
local t = {}
for _, key in ipairs({ 'alpha', 'beta', 'gamma', 'delta', 'epsilon', 'zeta', 'eta', 'theta', 'iota', 'kappa', 'lambda', 'mu' }) do
  t[key] = true
end
local order = {}
for key in pairs(t) do
  order[#order + 1] = key
end
io.stdout:write(table.concat(order, ' '), '\n')

==== script rm-pairs-probe.lua (the re-measure's driver; this pass changed 4 to 8 starts) ====
-- Runs rm-pairs-order.lua in eight fresh Neovims (isolated by make test_file)
-- and records the orders they print.
local MiniTest = require('mini.test')
local ROOT = vim.fn.getcwd()

local T = MiniTest.new_set()

T['pairs order'] = function()
  local lines = {}
  for _ = 1, 8 do
    local result = vim
      .system({ vim.v.progpath, '--clean', '--headless', '-i', 'NONE', '-l', ROOT .. '/.tests/rm-pairs-order.lua' })
      :wait()
    lines[#lines + 1] = vim.trim(result.stdout or '') .. vim.trim(result.stderr or '')
  end
  vim.fn.writefile(lines, ROOT .. '/.tests/rm-pairs-order.txt')
end

return T

==== script pairs-same-probe.lua (the test file's own Neovim builds the tables) ====
-- Builds the same twelve-key table five times in this one Neovim and records
-- each order pairs() visits it in, to see whether it varies within a process.
local MiniTest = require('mini.test')

local KEYS = { 'alpha', 'beta', 'gamma', 'delta', 'epsilon', 'zeta', 'eta', 'theta', 'iota', 'kappa', 'lambda', 'mu' }

local function order_of_a_new_table()
  local t = {}
  for _, key in ipairs(KEYS) do
    t[key] = true
  end
  local order = {}
  for key in pairs(t) do
    order[#order + 1] = key
  end
  return table.concat(order, ' ')
end

local T = MiniTest.new_set()

T['pairs order within one process'] = function()
  local lines = {}
  for _ = 1, 5 do
    lines[#lines + 1] = order_of_a_new_table()
    collectgarbage()
  end
  vim.fn.writefile(lines, vim.fn.getcwd() .. '/.tests/pairs-same-order.txt')
end

return T
