==== script probe-modes.lua ====
-- Which mode a child Neovim is in right after keys that enter Insert or
-- Terminal mode, by how the keys were given: pending (nvim_input, as
-- mini.test's child.type_keys gives them) or executed at once
-- (nvim_feedkeys with 'x', as a helper's press() gives them).
-- Each case runs in a fresh child: nvim --clean --headless --embed.

local function child()
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.keymap.set('n', '<Space>i', function() vim.cmd.startinsert() end)
    vim.keymap.set('n', '<Space>m', function()
      vim.cmd.startinsert()
      _G.mode_inside = vim.api.nvim_get_mode().mode
    end)
    vim.keymap.set('n', '<Space>w', function()
      vim.cmd.startinsert()
      vim.cmd.wincmd('w')
    end)
  ]], {})
  return chan
end

local function terminal(chan)
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.cmd.enew()
    vim.fn.jobstart({ 'cat' }, { term = true })
  ]], {})
end

local function state(chan)
  return vim.rpcrequest(chan, 'nvim_exec_lua', [[
    local buf = vim.api.nvim_get_current_buf()
    return ('mode=%s buftype=%s'):format(vim.api.nvim_get_mode().mode, vim.bo[buf].buftype == '' and 'normal' or vim.bo[buf].buftype)
  ]], {})
end

local function feed(chan, keys, flags)
  vim.rpcrequest(chan, 'nvim_exec_lua', 'vim.api.nvim_feedkeys(vim.keycode(...), select(2, ...), false)', { keys, flags })
end

local function input(chan, keys)
  vim.rpcrequest(chan, 'nvim_input', keys)
  vim.wait(200)
end

local cases = {
  { 'A normal buffer, <Space>i (startinsert in a mapping) typed as pending keys', function(c) input(c, '<Space>i') end },
  { 'B normal buffer, <Space>i fed with feedkeys "mx"', function(c) feed(c, '<Space>i', 'mx') end },
  { 'D normal buffer, i fed with feedkeys "nx"', function(c) feed(c, 'i', 'nx') end },
  { 'E terminal buffer, <Space>i typed as pending keys', function(c) terminal(c); input(c, '<Space>i') end },
  { 'F terminal buffer, <Space>i fed with feedkeys "mx"', function(c) terminal(c); feed(c, '<Space>i', 'mx') end },
  { 'G terminal buffer, i fed with feedkeys "nx"', function(c) terminal(c); feed(c, 'i', 'nx') end },
  { 'H terminal buffer, :startinsert run by nvim_command', function(c) terminal(c); vim.rpcrequest(c, 'nvim_command', 'startinsert'); vim.wait(200) end },
}

io.stdout:write(tostring(vim.version()) .. '\n')
for _, case in ipairs(cases) do
  local c = child()
  case[2](c)
  io.stdout:write(('%-78s -> %s\n'):format(case[1], state(c)))
  vim.fn.jobstop(c)
end

-- The mode read inside the callback that ran :startinsert.
local c = child()
input(c, '<Space>m')
io.stdout:write(('%-78s -> inside=%s after: %s\n'):format('I normal buffer, mode read inside the callback after :startinsert', vim.rpcrequest(c, 'nvim_exec_lua', 'return _G.mode_inside', {}), state(c)))

-- :startinsert, then the callback moves to another window: which window takes the mode?

  vim.cmd.vsplit()
  vim.cmd.enew()

local before = state(c)

-- What nvim__inspect_cell() reads in a headless child with no UI, by when
-- it is called: several reads in the request that makes the first call,
-- then reads in later requests. The cell under test is "[" of "[done]",
-- drawn by an extmark in a group with a foreground and bold.

local function child()
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.o.termguicolors = true
    vim.cmd('highlight ProbeStatus guifg=#b3f6c0 gui=bold')
    vim.cmd('highlight ProbeTime guifg=#9b9ea4')
    vim.api.nvim_buf_set_lines(0, 0, -1, false, { '09:05 [done] Task' })
    local ns = vim.api.nvim_create_namespace('probe')
    vim.api.nvim_buf_set_extmark(0, ns, 0, 0, { end_col = 5, hl_group = 'ProbeTime' })
    vim.api.nvim_buf_set_extmark(0, ns, 0, 6, { end_col = 12, hl_group = 'ProbeStatus' })

  vim.rpcrequest(chan, 'nvim_command', 'redraw')

  local out = {}
  for _, col in ipairs({ ... }) do

==== script probe-highlight.lua ====
-- What `:highlight default link G Title` does to a group G that something
-- was given first, and which default link `:highlight clear` restores.
-- Each case runs in a fresh child (--clean --headless --embed); the "given
-- first" command reaches it typed on the command line (nvim_input), as an
-- API command (nvim_command) or from Lua (vim.cmd in nvim_exec_lua).

local function child()
  return vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
end

local function definition(chan)
  return vim.inspect(vim.rpcrequest(chan, 'nvim_get_hl', 0, { name = 'G' }), { newline = ' ', indent = '' })
end

local function given(chan, how, command)
  if how == 'typed' then
    vim.rpcrequest(chan, 'nvim_input', ':' .. command .. '<CR>')
    vim.wait(200)
  elseif how == 'api' then
    vim.rpcrequest(chan, 'nvim_command', command)
  else
    vim.rpcrequest(chan, 'nvim_exec_lua', 'vim.cmd(...)', { command })
  end
end

local out = { tostring(vim.version()) }
local cases = {
  { 'G gui=NONE cterm=NONE', 'lua' },
  { 'G gui=NONE cterm=NONE', 'api' },
  { 'G guifg=#ff0000', 'lua' },
  { 'link G NONE', 'lua' },
  { 'link G NONE', 'api' },
  { 'link G NONE', 'typed' },
  { 'default link G DiagnosticOk', 'lua' },
}
for _, case in ipairs(cases) do
  local chan = child()
  given(chan, case[2], 'highlight ' .. case[1])
  local before = definition(chan)
  vim.rpcrequest(chan, 'nvim_command', 'highlight default link G Title')
  local after = definition(chan)
  vim.rpcrequest(chan, 'nvim_command', 'highlight clear')
  local cleared = definition(chan)
  out[#out + 1] = ('given first (%s) :highlight %s -> %s; then default link G Title -> %s; then :highlight clear -> %s'):format(
    case[2],
    case[1],
    before,
    after,
    cleared
  )
  vim.fn.jobstop(chan)
end
io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-system-exit.lua ====
-- When vim.system() reports the exit of a process that leaves a descendant
-- holding its output: `sh -c 'sleep 3 & exit 3'` exits at once with code 3,
-- while its `sleep 3` inherits sh's stdout and stderr.
-- Compared with vim.uv.spawn, whose output goes to a file.

local here = vim.fn.fnamemodify(arg[0], ':p:h')
local COMMAND = { 'sh', '-c', 'sleep 3 & exit 3' }
local out = { tostring(vim.version()) }
local function log(s)
  out[#out + 1] = s
end
local function ms_since(t0)
  return ('%.0f'):format((vim.uv.hrtime() - t0) / 1e6)
end

-- 1. vim.system with its output captured (the default).
local t0 = vim.uv.hrtime()
local result
vim.system(COMMAND, {}, function(r)
  result = r
  result.at = ms_since(t0)
end)
vim.wait(10000, function()
  return result ~= nil
end, 10)
log(('vim.system, output captured: on_exit after %s ms, code=%s'):format(result.at, result.code))

-- 2. vim.system with stdout and stderr not captured.
t0 = vim.uv.hrtime()
result = nil
vim.system(COMMAND, { stdout = false, stderr = false }, function(r)
  result = r
  result.at = ms_since(t0)
end)
vim.wait(10000, function()
  return result ~= nil
end, 10)
log(('vim.system, stdout = false, stderr = false: on_exit after %s ms, code=%s'):format(result.at, result.code))

-- 3. SystemObj:wait(500) on the captured form.
t0 = vim.uv.hrtime()
local waited = vim.system(COMMAND):wait(500)
log(('vim.system(...):wait(500): returned after %s ms, result=%s'):format(ms_since(t0), waited and ('code=' .. waited.code) or 'nil'))
vim.wait(4000)

-- 4. vim.uv.spawn, stdout and stderr to a file.
local path = here .. '/system-exit-spawn-output.txt'
local fd = assert(vim.uv.fs_open(path, 'w', 420))
t0 = vim.uv.hrtime()
local spawned
local handle = vim.uv.spawn('sh', { args = { '-c', 'sleep 3 & exit 3' }, stdio = { nil, fd, fd } }, function(code, signal)
  spawned = { at = ms_since(t0), code = code, signal = signal }
end)
vim.uv.fs_close(fd)
vim.wait(10000, function()
  return spawned ~= nil
end, 10)
handle:close()
log(('vim.uv.spawn, output to a file: exit callback after %s ms, code=%s'):format(spawned.at, spawned.code))
vim.wait(3500)

io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-log-fallback.lua ====
-- What a Neovim whose NVIM_LOG_FILE cannot be opened hands to the Neovims it
-- starts. Run with NVIM_LOG_FILE naming a file in a directory that does not
-- exist. Each child is nvim --clean --headless --embed; its messages are read
-- 400 ms after it started (the notice is deferred by 100 ms after VimEnter).

local here = vim.fn.fnamemodify(arg[0], ':p:h')
local function short(value)
  if value == nil or value == vim.NIL then
    return 'unset'
  end
  return (tostring(value):gsub(vim.pesc(here), '<scratch>'))
end

local out = { tostring(vim.version()) }
out[#out + 1] = ('parent: NVIM_LOG_FILE=%s __NVIM_LOG_FILE_WANT=%s'):format(short(vim.env.NVIM_LOG_FILE), short(vim.env.__NVIM_LOG_FILE_WANT))

local function child(label, env)
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true, env = env })
  vim.wait(400)
  local seen = vim.rpcrequest(chan, 'nvim_exec_lua', [[
    return {
      log = vim.env.NVIM_LOG_FILE,
      want = vim.env.__NVIM_LOG_FILE_WANT,
      messages = vim.api.nvim_exec2('messages', { output = true }).output,
    }
  ]], {})
  out[#out + 1] = ('%s: NVIM_LOG_FILE=%s __NVIM_LOG_FILE_WANT=%s messages=%s'):format(
    label,
    short(seen.log),
    short(seen.want),
    seen.messages == '' and '(none)' or vim.inspect(short(seen.messages))
  )
  vim.fn.jobstop(chan)
end

vim.fn.mkdir(here .. '/child-log', 'p')
child('child, environment inherited', nil)
child('child, its own NVIM_LOG_FILE in a directory that exists', { NVIM_LOG_FILE = here .. '/child-log/log' })

io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-jobs.lua ====
-- 1. jobwait({id}, 0) and vim.uv.kill(pid, 0) on a job jobstop() has stopped
--    but that still runs: it ignores SIGHUP and SIGTERM.
-- 2. The order of TermClose and the job's on_exit when a terminal job ends.

local DEAF = { 'sh', '-c', 'trap "" HUP TERM; while :; do sleep 1; done' }
local out = { tostring(vim.version()) }
local function log(s)
  out[#out + 1] = s
end
local function ms_since(t0)
  return ('%.0f'):format((vim.uv.hrtime() - t0) / 1e6)
end

local function stopped_job(label, start)
  local id = start()
  local pid = vim.fn.jobpid(id)
  vim.wait(300)
  vim.fn.jobstop(id)
  local t0 = vim.uv.hrtime()
  local alive = vim.uv.kill(pid, 0)
  log(('%s: vim.uv.kill(pid, 0) right after jobstop() returned %s in %s ms'):format(label, tostring(alive), ms_since(t0)))
  t0 = vim.uv.hrtime()
  local status = vim.fn.jobwait({ id }, 0)[1]
  log(('%s: jobwait({id}, 0) returned %d after %s ms'):format(label, status, ms_since(t0)))
  local gone, err = vim.uv.kill(pid, 0)
  log(('%s: vim.uv.kill(pid, 0) afterwards returned %s %s'):format(label, tostring(gone), tostring(err)))
end

stopped_job('pipe job', function()
  return vim.fn.jobstart(DEAF)
end)
stopped_job('terminal job', function()
  vim.cmd.enew()
  return vim.fn.jobstart(DEAF, { term = true })
end)

for run = 1, 5 do
  local order = {}
  vim.cmd.enew()
  local buf = vim.api.nvim_get_current_buf()
  vim.api.nvim_create_autocmd('TermClose', {
    buffer = buf,
    once = true,
    callback = function()
      order[#order + 1] = 'TermClose'
    end,
  })
  vim.fn.jobstart({ 'sh', '-c', 'exit 3' }, {
    term = true,
    on_exit = function()
      order[#order + 1] = 'on_exit'
    end,
  })
  vim.wait(3000, function()
    return #order == 2
  end, 10)
  log(('terminal job exiting by itself, run %d: %s'):format(run, table.concat(order, ' then ')))
end

io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-bwipeout.lua ====
-- When :bwipeout! of a terminal buffer shown in the current window puts
-- another buffer in that window: before BufWipeout or after it, for a
-- terminal whose job still runs and for one whose job has ended. The other
-- window shows an unlisted scratch buffer, so no listed buffer is left.

local out = { tostring(vim.version()) }

local function trace(label, job_ends_first)
  vim.cmd('silent! %bwipeout!')
  vim.cmd.enew()
  local scratch = vim.api.nvim_get_current_buf()
  vim.bo[scratch].buflisted = false
  vim.bo[scratch].buftype = 'nofile'
  vim.cmd.vsplit()
  vim.cmd.enew()
  local win = vim.api.nvim_get_current_win()
  local job = vim.fn.jobstart(job_ends_first and { 'sh', '-c', 'exit 0' } or { 'cat' }, { term = true })
  local term = vim.api.nvim_get_current_buf()
  vim.bo[term].buflisted = true
  if job_ends_first then
    vim.fn.jobwait({ job }, 2000)
    vim.wait(200)
  end
  local events = {}
  local group = vim.api.nvim_create_augroup('probe_bwipeout', {})
  for _, event in ipairs({ 'TermClose', 'BufUnload', 'BufWipeout', 'BufWinEnter', 'BufEnter' }) do
    vim.api.nvim_create_autocmd(event, {
      group = group,
      callback = function(args)
        local shown = vim.api.nvim_win_is_valid(win) and vim.api.nvim_win_get_buf(win)
        events[#events + 1] = ('%s(buf %s; the window shows %s)'):format(
          event,
          args.buf == term and 'term' or tostring(args.buf),
          shown == term and 'term' or (shown and ('buf ' .. shown) or 'nothing: closed')
        )
      end,
    })
  end
  vim.cmd('bwipeout!')
  vim.api.nvim_del_augroup_by_id(group)
  out[#out + 1] = ('%s: %s'):format(label, table.concat(events, ' -> '))
  out[#out + 1] = ('%s: afterwards the window %s'):format(
    label,
    vim.api.nvim_win_is_valid(win)
        and ('shows buf %d, name %q, listed %s'):format(
          vim.api.nvim_win_get_buf(win),
          vim.api.nvim_buf_get_name(vim.api.nvim_win_get_buf(win)),
          tostring(vim.bo[vim.api.nvim_win_get_buf(win)].buflisted)
        )
      or 'is closed'
  )
  vim.cmd('silent! only')
end

trace('running terminal', false)
trace('ended terminal', true)

==== script keys-reader.py (the program in the terminal) ====
"""Runs inside a Neovim terminal: optionally enables the kitty keyboard
protocol with the given flags (CSI > flags u), asks for the flags in force
(CSI ? u), then records every byte its terminal sends for a few seconds.
Usage: keys-reader.py <out file> <flags or 'none'>"""
import os
import select
import sys
import termios
import time
import tty

out_path, flags = sys.argv[1], sys.argv[2]
fd = sys.stdin.fileno()
tty.setraw(fd)
if flags != 'none':
    os.write(sys.stdout.fileno(), ('\x1b[>%su' % flags).encode())
    os.write(sys.stdout.fileno(), b'\x1b[?u')
os.write(sys.stdout.fileno(), b'ready\r\n')
received = b''
deadline = time.time() + 4
while time.time() < deadline:
    ready, _, _ = select.select([fd], [], [], 0.1)
    if ready:
        received += os.read(fd, 1024)
with open(out_path, 'w') as out:
    out.write(repr(received))

-- When a child Neovim enters Insert or Terminal mode for keys fed with
-- nvim_feedkeys(): inside the call, or after the request that made it.
-- Each case runs in a fresh child: nvim --clean --headless --embed. One
-- request feeds the keys and reads the mode before it returns ("same
-- request"); the next request reads the mode again ("next request"). A
-- ModeChanged autocommand logs every change as old>new, marked "(in call)"
-- when it fired while nvim_feedkeys() was running.

local function child()
  local chan = vim.fn.jobstart({ vim.v.progpath, '--clean', '--headless', '--embed' }, { rpc = true })
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.keymap.set('n', '<Space>i', function() vim.cmd.startinsert() end)
    _G.changes = {}
    _G.in_call = false
    vim.api.nvim_create_autocmd('ModeChanged', {
      callback = function()
        local e = vim.v.event
        _G.changes[#_G.changes + 1] = e.old_mode .. '>' .. e.new_mode .. (_G.in_call and '(in call)' or '')
      end,
    })
  ]], {})
  return chan
end

local function terminal(chan)
  vim.rpcrequest(chan, 'nvim_exec_lua', [[
    vim.cmd.enew()
    vim.fn.jobstart({ 'cat' }, { term = true })
    _G.changes = {}
  ]], {})
end

  local keys, flags = ...
  _G.in_call = true
  vim.api.nvim_feedkeys(vim.keycode(keys), flags, false)
  _G.in_call = false
  return vim.api.nvim_get_mode().mode
]]

-- signals itself, and one that vim.system's own timeout ends (:wait's, which
-- sends SIGKILL, and the timeout option's, which sends SIGTERM).
local out = { tostring(vim.version()) }
for _, sig in ipairs({ 'KILL', 'TERM', 'SEGV' }) do
  local r = vim.system({ 'sh', '-c', 'kill -' .. sig .. ' $$' }):wait(5000)
  out[#out + 1] = ('sh kills itself with %s: code=%s signal=%s'):format(sig, tostring(r and r.code), tostring(r and r.signal))
end
local waited = vim.system({ 'sleep', '5' }):wait(200)
out[#out + 1] = ('sleep 5, ended by :wait(200): code=%s signal=%s'):format(tostring(waited and waited.code), tostring(waited and waited.signal))
local timed = vim.system({ 'sleep', '5' }, { timeout = 200 }):wait(5000)
out[#out + 1] = ('sleep 5, ended by { timeout = 200 }: code=%s signal=%s'):format(tostring(timed and timed.code), tostring(timed and timed.signal))
io.stdout:write(table.concat(out, '\n') .. '\n')

==== script probe-softwrap.lua ====
-- Records probe: rows a Neovim terminal holds when a program prints the
-- no-conversation message as one line and the terminal soft-wraps it.
-- The real claude is never run; printf stands in for a program that does not
-- break its own lines.
local message = 'No conversation found with session ID: 11111111-2222-4333-8444-555555555555'
local out = { tostring(vim.version()) }
for _, width in ipairs({ 39, 60 }) do
  vim.cmd('silent! only')
  vim.o.columns = 200
  vim.cmd('vsplit')
  vim.cmd('enew')
  vim.api.nvim_win_set_width(0, width)
  local buf = vim.api.nvim_get_current_buf()
  local done = false
  vim.fn.jobstart({ 'printf', '%s', message }, {
    term = true,
    on_exit = function()
      done = true
    end,
  })
  vim.wait(3000, function()
    return done
  end, 20)
  vim.wait(200)
  out[#out + 1] = ('width %d, window %d:'):format(width, vim.api.nvim_win_get_width(0))
  for i, l in ipairs(vim.api.nvim_buf_get_lines(buf, 0, -1, false)) do
    if l ~= '' then
      out[#out + 1] = ('%02d|%s|'):format(i, l)
    end
  end
end
io.stdout:write(table.concat(out, '\n') .. '\n')
